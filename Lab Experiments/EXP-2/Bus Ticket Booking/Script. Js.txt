/* =========================================

function money(number) {

    return "₹" + number.toLocaleString("en-IN");

}


function showToast(message) {

    toast.textContent = message;

    toast.classList.add("show");

    setTimeout(() => {

        toast.classList.remove("show");

    }, 2500);

}


function showPage(page) {

    state.page = page;

function toggleSeat(seat) {

if (state.selectedSeats.includes(seat)) {

state.selectedSeats =
state.selectedSeats.filter(
item => item !== seat
);

}

else {

if (
state.selectedSeats.length >=
Number(state.search.passengers)
) {

showToast(
"Select only " +
state.search.passengers +
" seat(s)."
);

return;

}

state.selectedSeats.push(seat);

}

render();

}


function continuePassenger() {

if (
state.selectedSeats.length !==
Number(state.search.passengers)
) {

showToast(
"Please select " +
state.search.passengers +
" seat(s)."
);

return;

}

showPage("passenger");

}

function applyCoupon() {

const code =
document
.getElementById("coupon")
.value
.trim()
.toUpperCase();


if (code === "FIRSTGO") {

state.coupon = 150;

showToast("₹150 discount applied!");

}

else if (code === "WEEKEND20") {

state.coupon =
Math.round(
state.selectedBus.price *
state.selectedSeats.length *
0.20
);

showToast("20% discount applied!");

}

else {

state.coupon = 0;

showToast("Invalid coupon.");

}

render();

}


/* =========================================
   PAYMENT
   ========================================= */

function payNow() {

const bus = state.selectedBus;

const total =
Math.max(
0,
bus.price *
state.selectedSeats.length +
20 -
state.coupon
);


state.booking = {

id:
"BG" +
Date.now()
.toString()
.slice(-8),

date:
state.search.date,

total: total,

seats:
[...state.selectedSeats],

passenger:
state.passenger,

bus:
bus,

payment:
state.payment

};


localStorage.setItem(
"busgoBooking",
JSON.stringify(state.booking)
);


showToast(
"Payment successful!"
);


setTimeout(() => {

showPage("confirmation");

}, 700);

}

/* =========================================
   DOWNLOAD TICKET
   ========================================= */

function downloadTicket() {

const booking = state.booking;

const text = `

BUSGO E-TICKET

Booking ID:
${booking.id}

Route:
${booking.bus.from} → ${booking.bus.to}

Travel Date:
${booking.date}

Bus:
${booking.bus.operator}

Departure:
${booking.bus.depart}

Seat:
${booking.seats.join(", ")}

Passenger:
${booking.passenger.name}

Fare:
${money(booking.total)}

Status:
CONFIRMED

Thank you for booking with BusGo!

`;


const blob =
new Blob(
[text],
{
type: "text/plain"
}
);


const url =
URL.createObjectURL(blob);


const a =
document.createElement("a");


a.href = url;

a.download =
`BusGo-${booking.id}.txt`;


a.click();


URL.revokeObjectURL(url);

}

homePage();

}

else if (state.page === "search") {

app.innerHTML =
searchPage();

}

else if (state.page === "seats") {

app.innerHTML =
seatsPage();

}

else if (state.page === "passenger") {

app.innerHTML =
passengerPage();

}

else if (state.page === "payment") {

app.innerHTML =
paymentPage();

}

else if (state.page === "confirmation") {

app.innerHTML =
confirmationPage();

}

else if (state.page === "bookings") {

app.innerHTML =
bookingsPage();

state.page === "login" ||
state.page === "signup"
) {

authPage(state.page);

}


/* =========================================
   EVENT HANDLERS
   ========================================= */

function attachEvents() {


/* SEARCH FORM */

const searchForm =
document.getElementById(
"searchForm"
);


if (searchForm) {

searchForm.onsubmit =
function(event) {

event.preventDefault();


state.search = {

from:
document.getElementById(
"from"
).value,

to:
document.getElementById(
"to"
).value,

date:
document.getElementById(
"date"
).value,

passengers:
Number(
document.getElementById(
"passengers"
).value
)

};


if (
state.search.from ===
state.search.to
) {

showToast(
"From and To cities must be different."
);

return;

}


showPage("search");

};

}


/* PASSENGER FORM */

const passengerForm =
document.getElementById(
"passengerForm"
);


if (passengerForm) {

passengerForm.onsubmit =
function(event) {

event.preventDefault();


state.passenger = {

name:
document.getElementById(
"name"
).value,

age:
document.getElementById(
"age"
).value,

gender:
document.getElementById(
"gender"
).value,

email:
document.getElementById(
"email"
).value,

phone:
document.getElementById(
"phone"
).value,

boarding:
document.getElementById(
"boarding"
).value,

dropping:
document.getElementById(
"dropping"
).value

};


showPage("payment");

};

}


/* LOGIN */

const authForm =
document.getElementById(
"authForm"
);


if (authForm) {

authForm.onsubmit =
function(event) {

event.preventDefault();


showToast(
state.page === "login"
?
"Login successful!"
:
"Account created successfully!"
);


setTimeout(() => {

showPage("home");

}, 700);

};

}


/* SORTING */

const sortSelect =
document.getElementById(
"sortSelect"
);


if (sortSelect) {

sortSelect.onchange =
function() {

let results =
buses.filter(bus =>

bus.from === state.search.from &&
bus.to === state.search.to

);


if (
this.value === "price"
) {

results.sort(
(a,b) =>
a.price - b.price
);

}


if (
this.value === "duration"
) {

results.sort(
(a,b) =>
parseInt(a.duration) -
parseInt(b.duration)
);

}


if (
this.value === "departure"
) {

results.sort(
(a,b) =>
a.depart.localeCompare(
b.depart
)
);

}


const section =
document.querySelector(
".results-layout section"
);


section.innerHTML = `

<div class="sortbar">

<b>
${results.length}
Buses Found
</b>

<select id="sortSelect">

<option value="recommended">
Recommended
</option>

<option value="price">
Price: Low to High
</option>

<option value="duration">
Duration
</option>

<option value="departure">
Departure Time
</option>

</select>

</div>

${results.map(busCard).join("")}

`;


attachEvents();

};

}

}
