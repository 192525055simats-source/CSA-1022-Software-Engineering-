// ================= PAGE NAVIGATION =================

function showPage(pageId) {

    document.querySelectorAll(".page").forEach(page => {
        page.classList.remove("active");
    });

    const page = document.getElementById(pageId);

    if (page) {
        page.classList.add("active");
    }

    window.scrollTo({
        top: 0,
        behavior: "smooth"
    });

    if (pageId === "cart") {
        renderCart();
    }

    if (pageId === "checkout") {
        renderCheckout();
    }

    if (pageId === "wishlist") {
        renderWishlist();
    }

    if (pageId === "orders") {
        updateCartCount();
    }

    if (pageId === "payment") {
        updatePaymentSummary();
    }

}

// ================= RENDER PRODUCTS =================

function renderProducts() {

        filteredProducts.map(productCard).join("");

// ================= SEARCH =================

function searchProducts() {

    const keyword =
        document.getElementById("searchInput")
            .value
            .toLowerCase();

    filteredProducts = products.filter(product =>

        product.name.toLowerCase().includes(keyword) ||

        product.brand.toLowerCase().includes(keyword) ||

        product.category.toLowerCase().includes(keyword)

    );

    document.getElementById("shopProducts").innerHTML =
        filteredProducts.map(productCard).join("");

    document.getElementById("productCount").textContent =
        `${filteredProducts.length} Products found`;

    if (keyword.length > 0) {
        showPage("shop");
    }

}


// ================= CATEGORY FILTER =================

function filterCategory(category) {

    filteredProducts =
        products.filter(product =>
            product.category === category
        );

    renderProducts();

    showPage("shop");

}


// ================= PRICE FILTER =================

function filterPrice() {

    const value =
        Number(document.getElementById("priceRange").value);

    document.getElementById("priceValue").textContent =
        value.toLocaleString();

    filteredProducts =
        products.filter(product =>
            product.price <= value
        );

    document.getElementById("shopProducts").innerHTML =
        filteredProducts.map(productCard).join("");

}


// ================= SORT =================

function sortProducts() {

    const option =
        document.getElementById("sortSelect").value;

    if (option === "Price: Low to High") {

        filteredProducts.sort(
            (a, b) => a.price - b.price
        );

    }

    if (option === "Price: High to Low") {

        filteredProducts.sort(
            (a, b) => b.price - a.price
        );

    }

    if (option === "New Arrivals") {

        filteredProducts.reverse();

    }

    document.getElementById("shopProducts").innerHTML =
        filteredProducts.map(productCard).join("");

}


// ================= PRODUCT DETAILS =================

function openProduct(id) {

    const product =
        products.find(item => item.id === id);

    if (!product) return;

    currentProduct = product;

    quantity = 1;

    document.getElementById("detailName").textContent =
        product.name;

    document.getElementById("detailBrand").textContent =
        product.brand;

    document.getElementById("detailPrice").textContent =
        "₹" + product.price.toLocaleString();

    document.getElementById("detailOldPrice").textContent =
        "₹" + product.oldPrice.toLocaleString();

    document.getElementById("detailImage").textContent =
        product.icon;

    document.getElementById("quantity").textContent =
        quantity;

    showPage("productDetails");

}


// ================= QUANTITY =================

function changeQuantity(value) {

    quantity += value;

    if (quantity < 1) {
        quantity = 1;
    }

    if (quantity > 10) {
        quantity = 10;
    }

    document.getElementById("quantity").textContent =
        quantity;

}


// ================= CART =================

function addToCartById(id) {

    const product =
        products.find(item => item.id === id);

    addToCart(product);

}


function addToCart(product) {

    if (!product) return;

    const existing =
        cart.find(item => item.id === product.id);

    if (existing) {

        existing.quantity++;

    } else {

        cart.push({
            ...product,
            quantity: 1
        });

    }

    saveCart();

    updateCartCount();

    alert(
        product.name +
        " added to your cart!"
    );

}


function addCurrentProduct() {

    for (let i = 0; i < quantity; i++) {
        addToCart(currentProduct);
    }

    showPage("cart");

}


function buyNow() {

    cart = [];

    for (let i = 0; i < quantity; i++) {
        cart.push({
            ...currentProduct,
            quantity: 1
        });
    }

    saveCart();

    showPage("checkout");

}


// ================= CART RENDER =================

function renderCart() {

    const container =
        document.getElementById("cartItems");

    if (cart.length === 0) {

        container.innerHTML = `

            <div class="checkout-card">

                <h2>Your cart is empty</h2>

                <p>
                    Start shopping and add products
                    to your cart.
                </p>

                <br>

                <button
                    class="primary-btn"
                    onclick="showPage('shop')"
                >
                    Continue Shopping
                </button>

            </div>

        `;

        updateCartSummary();

        return;
    }


    container.innerHTML =
        cart.map(item => `

            <div class="cart-item">

                <div class="cart-item-image">
                    ${item.icon}
                </div>

                <div class="cart-item-info">

                    <span class="brand">
                        ${item.brand}
                    </span>

                    <h3>${item.name}</h3>

                    <p>
                        ₹${item.price.toLocaleString()}
                    </p>

                    <button
                        class="remove-btn"
                        onclick="removeFromCart(${item.id})"
                    >
                        Remove
                    </button>

                </div>


                <div class="cart-controls">

                    <button
                        onclick="changeCartQuantity(${item.id}, -1)"
                    >
                        −
                    </button>

                    <span>${item.quantity}</span>

                    <button
                        onclick="changeCartQuantity(${item.id}, 1)"
                    >
                        +
                    </button>

                </div>


                <strong>
                    ₹${(
                        item.price * item.quantity
                    ).toLocaleString()}
                </strong>

            </div>

        `).join("");

    updateCartSummary();

}


// ================= CART QUANTITY =================

function changeCartQuantity(id, amount) {

    const item =
        cart.find(product => product.id === id);

    if (!item) return;

    item.quantity += amount;

    if (item.quantity <= 0) {

        cart =
            cart.filter(product =>
                product.id !== id
            );

    }

    saveCart();

    renderCart();

}


// ================= REMOVE =================

function removeFromCart(id) {

    cart =
        cart.filter(product =>
            product.id !== id
        );

    saveCart();

    renderCart();

    updateCartCount();

}


// ================= SUMMARY =================

function getCartTotal() {

    return cart.reduce(
        (total, item) =>
            total + item.price * item.quantity,
        0
    );

}


function updateCartSummary() {

    const subtotal =
        getCartTotal();

    const discount =
        cart.reduce(
            (total, item) =>
                total +
                (
                    (item.oldPrice - item.price)
                    * item.quantity
                ),
            0
        );

    document.getElementById("subtotal").textContent =
        "₹" + subtotal.toLocaleString();

    document.getElementById("cartDiscount").textContent =
        "-₹" + discount.toLocaleString();

    document.getElementById("deliveryFee").textContent =
        subtotal >= 999 ? "FREE" : "₹99";

    document.getElementById("cartTotal").textContent =
        "₹" +
        (
            subtotal +
            (subtotal >= 999 ? 0 : 99)
        ).toLocaleString();

}


// ================= CART COUNT =================

function updateCartCount() {

    const count =
        cart.reduce(
            (total, item) =>
                total + item.quantity,
            0
        );

    document.getElementById("cartCount").textContent =
        count;

}

// ================= WISHLIST =================

function toggleWishlist(id) {

    const product =
        products.find(item => item.id === id);

    const exists =
        wishlist.some(item => item.id === id);

    if (exists) {

        wishlist =
            wishlist.filter(item =>
                item.id !== id
            );

    } else {

        wishlist.push(product);

    }

    localStorage.setItem(
        "novamartWishlist",
        JSON.stringify(wishlist)
    );

    renderProducts();

    renderWishlist();

}


function addToWishlist(product) {

    if (!wishlist.some(item => item.id === product.id)) {

        wishlist.push(product);

        localStorage.setItem(
            "novamartWishlist",
            JSON.stringify(wishlist)
        );

        alert(
            "Added to Wishlist!"
        );

    }

}


function renderWishlist() {

    const container =
        document.getElementById("wishlistProducts");

    if (wishlist.length === 0) {

        container.innerHTML = `

            <div class="checkout-card">

                <h2>Your Wishlist is Empty</h2>

                <p>
                    Save products you love here.
                </p>

            </div>

        `;

        return;
    }

    container.innerHTML =
        wishlist.map(productCard).join("");

}
